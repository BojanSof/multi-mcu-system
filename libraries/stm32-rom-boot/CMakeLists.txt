cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(Stm32RomBoot LANGUAGES C)
    include(CTest)
endif()
include(cmake/Sources.cmake)
add_library(stm32_rom_boot STATIC ${STM32_ROM_BOOT_SOURCES})
add_library(Stm32RomBoot::Stm32RomBoot ALIAS stm32_rom_boot)
target_include_directories(stm32_rom_boot PUBLIC include)
target_compile_features(stm32_rom_boot PUBLIC c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(stm32_rom_boot PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
endif()
if(BUILD_TESTING)
    add_executable(stm32_rom_boot_test tests/Stm32RomBootTest.c)
    target_link_libraries(stm32_rom_boot_test PRIVATE Stm32RomBoot::Stm32RomBoot)
    target_compile_options(stm32_rom_boot_test PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror -UNDEBUG)
    add_test(NAME stm32_rom_boot.transactions COMMAND stm32_rom_boot_test)
endif()
