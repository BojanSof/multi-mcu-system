cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(Protocol LANGUAGES C)
    include(CTest)
endif()
if(NOT COMMAND multi_mcu_apply_warnings)
    function(multi_mcu_apply_warnings target)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
    endfunction()
endif()
include(cmake/Sources.cmake)
add_library(protocol STATIC ${PROTOCOL_SOURCES})
add_library(Protocol::Protocol ALIAS protocol)
target_include_directories(protocol PUBLIC include)
target_compile_features(protocol PUBLIC c_std_11)
multi_mcu_apply_warnings(protocol)
if(BUILD_TESTING)
    add_executable(protocol_test tests/ProtocolTest.c)
    target_link_libraries(protocol_test PRIVATE Protocol::Protocol)
    multi_mcu_apply_warnings(protocol_test)
    add_test(NAME protocol.version COMMAND protocol_test)
endif()
