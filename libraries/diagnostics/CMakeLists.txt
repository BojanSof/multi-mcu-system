cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(Diagnostics LANGUAGES C)
    include(CTest)
endif()
if(NOT COMMAND multi_mcu_apply_warnings)
    function(multi_mcu_apply_warnings target)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
    endfunction()
endif()
include(cmake/Sources.cmake)
add_library(diagnostics STATIC ${DIAGNOSTICS_SOURCES})
add_library(Diagnostics::Diagnostics ALIAS diagnostics)
target_include_directories(diagnostics PUBLIC include)
target_compile_features(diagnostics PUBLIC c_std_11)
multi_mcu_apply_warnings(diagnostics)
if(BUILD_TESTING)
    add_executable(diagnostics_test tests/DiagnosticsTest.c)
    target_link_libraries(diagnostics_test PRIVATE Diagnostics::Diagnostics)
    multi_mcu_apply_warnings(diagnostics_test)
    add_test(NAME diagnostics.counter COMMAND diagnostics_test)
endif()
