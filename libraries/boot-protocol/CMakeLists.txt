cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(BootProtocol LANGUAGES C)
    include(CTest)
endif()
if(NOT COMMAND multi_mcu_apply_warnings)
    function(multi_mcu_apply_warnings target)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
    endfunction()
endif()
include(cmake/Sources.cmake)
add_library(boot_protocol STATIC ${BOOT_PROTOCOL_SOURCES})
add_library(BootProtocol::BootProtocol ALIAS boot_protocol)
target_include_directories(boot_protocol PUBLIC include)
target_compile_features(boot_protocol PUBLIC c_std_11)
multi_mcu_apply_warnings(boot_protocol)
if(BUILD_TESTING)
    add_executable(boot_protocol_test tests/BootProtocolTest.c)
    target_link_libraries(boot_protocol_test PRIVATE BootProtocol::BootProtocol)
    multi_mcu_apply_warnings(boot_protocol_test)
    add_test(NAME boot_protocol.frames COMMAND boot_protocol_test)
endif()
