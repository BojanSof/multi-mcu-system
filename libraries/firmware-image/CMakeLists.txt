cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(FirmwareImage LANGUAGES C)
    include(CTest)
endif()
if(NOT COMMAND multi_mcu_apply_warnings)
    function(multi_mcu_apply_warnings target)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
    endfunction()
endif()
include(cmake/Sources.cmake)
add_library(firmware_image STATIC ${FIRMWARE_IMAGE_SOURCES})
add_library(FirmwareImage::FirmwareImage ALIAS firmware_image)
target_include_directories(firmware_image PUBLIC include)
target_compile_features(firmware_image PUBLIC c_std_11)
multi_mcu_apply_warnings(firmware_image)
if(BUILD_TESTING)
    add_executable(firmware_image_test tests/FirmwareImageTest.c)
    target_link_libraries(firmware_image_test PRIVATE FirmwareImage::FirmwareImage)
    multi_mcu_apply_warnings(firmware_image_test)
    add_test(NAME firmware_image.manifest COMMAND firmware_image_test)
endif()
