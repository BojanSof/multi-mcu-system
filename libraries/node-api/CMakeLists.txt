cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(NodeApi LANGUAGES C)
    include(CTest)
endif()
if(NOT COMMAND multi_mcu_apply_warnings)
    function(multi_mcu_apply_warnings target)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
    endfunction()
endif()
include(cmake/Sources.cmake)
add_library(node_api STATIC ${NODE_API_SOURCES})
add_library(NodeApi::NodeApi ALIAS node_api)
target_include_directories(node_api PUBLIC include)
target_compile_features(node_api PUBLIC c_std_11)
multi_mcu_apply_warnings(node_api)
if(BUILD_TESTING)
    add_executable(node_api_test tests/NodeApiTest.c)
    target_link_libraries(node_api_test PRIVATE NodeApi::NodeApi)
    multi_mcu_apply_warnings(node_api_test)
    add_test(NAME node_api.version COMMAND node_api_test)
    add_executable(node_events_test tests/NodeReliableEventsTest.c)
    target_link_libraries(node_events_test PRIVATE NodeApi::NodeApi)
    multi_mcu_apply_warnings(node_events_test)
    add_test(NAME node_api.reliable_events COMMAND node_events_test)
endif()
