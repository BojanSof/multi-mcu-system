cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(VirtualIo LANGUAGES C)
    include(CTest)
    add_subdirectory(../../libraries/node-api node-api)
endif()
include(cmake/Sources.cmake)
if(NOT COMMAND multi_mcu_apply_warnings)
    function(multi_mcu_apply_warnings target)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
    endfunction()
endif()
add_library(virtual_io STATIC ${VIRTUAL_IO_SOURCES})
add_library(VirtualIo::VirtualIo ALIAS virtual_io)
target_include_directories(virtual_io PUBLIC include)
target_link_libraries(virtual_io PUBLIC NodeApi::NodeApi)
target_compile_features(virtual_io PUBLIC c_std_11)
multi_mcu_apply_warnings(virtual_io)
if(BUILD_TESTING)
    add_executable(virtual_io_test tests/VirtualIoTest.c)
    target_link_libraries(virtual_io_test PRIVATE VirtualIo::VirtualIo)
    multi_mcu_apply_warnings(virtual_io_test)
    add_test(NAME virtual_io.behavior COMMAND virtual_io_test)
endif()
